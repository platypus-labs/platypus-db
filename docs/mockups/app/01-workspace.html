1-- Fatturato per città, ultimi 30 giorni
2SELECT c.city, count(*) AS orders, sum(o.total) AS revenue
3FROM orders o
4JOIN customers c ON c.id = o.customer_id
5WHERE o.placed_at > now() - interval '30 days'
6GROUP BY c.city ORDER BY revenue DESC;
